#  Create mask (removes horizontal stripes)
mask = np.ones((height, width), dtype=np.float32)

Y, X = np.ogrid[:height, :width]
In [56]:
#  Horizontal Moiré: remove two symmetric frequencies
notch_radius = 18
notches = [
    # below center
    (height//2 + 55, width//2),
     # above center
    (height//2 - 55, width//2),
]

for cy, cx in notches:
    mask[(Y - cy)**2 + (X - cx)**2 < notch_radius**2] = 0
In [57]:
# Remove vertical band by notch line
vertical_band_width = 12
# shift from center
cx = width//2 + 60
mask[:, cx-vertical_band_width:cx+vertical_band_width] = 0
In [58]:
#  Apply mask
F_filtered = shifted_fourier * mask

# Inverse FFT
restored = np.abs(ifft2(ifftshift(F_filtered)))
